class Location {
    private double latitude;
    private double longitude;
    private double currentBusSpeed;

    public Location(double latitude, double longitude, double speed) {
        this.latitude = latitude;
        this.longitude = longitude;
        this.currentBusSpeed = speed;
    }

    public double getLatitude() {
        return latitude; 
    }
    public double getLongitude() {
        return longitude;
    }
}

class LocationService {
	private ArrayList<Location> locations = new ArrayList<>();

    public Location getCurrentLocation() {
        int numLocations = locations.size();
        return locations.get(numLocations - 1);
    }
}

public class BusGuideanceController {
    private BusConsoleUI ui = new BusConsoleUI();
    private LocationService locationService = new LocationService();
    private Database db = new Database();
    private Report activeReport;

    public void initialiseReport(BusDriver driver, String reportMethod) {
        if (!driver.isAuthenticated()) {
            return;
        }

        if (reportMethod.equals("Voice")) {
            ui.listenForSpeech();
            String parsedText = ui.processSpeechToText();
            ui.displayParsedText(parsedText);
            
            // IMPLEMENT IN SEQUENCE: OBJECT?
            String reportType = ui.getParsedReportType();
        } else {
            String reportType = ui.selectReportType();
        }

        Location currentLocation = locationService.getCurrentLocation();
        String timeStamp = getCurrentTime();

        if (reportType.equals("Hazard")) {
            activeReport = new HazardReport(currentLocation, timeStamp);
        } else {
            activeReport = new DelayReport(currentLocation, timeStamp);
        }

        db.submitReport(activeReport);
        String status = db.getDatabaseStatus();
        // escalateReport
        // publishIncidentToAPI

        if (status.equals("Success")) {
            ui.displaySuccessNotification();
            ui.audibleConfirmation("Report Sent Successfully");
        } else {
            ui.displayRetryNotification();
            ui.audibleRetryAlert("Report Failure, please try again.");
        }
    }
    
    private String getCurrentTime() {
        return "";
    }
